/** * SONG REQUEST → YOUTUBE PLAYLIST AUTOMATION * DJ Sunny — PartyQueue System * * WHAT THIS DOES: * • When a guest submits a song request via Google Form, * this script automatically searches YouTube for that song * and adds it to a daily playlist titled "MM/DD/YY". * • A new playlist is created each day automatically. * • The Google Sheet is updated with status, video ID, and playlist ID. * * HOW TO INSTALL / REINSTALL: * 1. Open your Google Sheet (the one connected to the request form) * 2. Go to Extensions → Apps Script * 3. Select ALL existing code (Cmd+A) and DELETE it * 4. Paste this entire script * 5. Click Save (Cmd+S) * 6. In the function dropdown (top bar), select "initialSetup" * 7. Click the ▶ Run button * 8. Authorize when prompted (allow YouTube + Sheets access) * 9. You should see "Setup complete!" in the execution log * * THAT'S IT. The script will now run automatically on every form submission. */ // ==================== CONFIGURATION ==================== const CONFIG = { PLAYLIST_PREFIX: 'Song Requests', // Column headers in your Google Sheet — adjust if yours differ COLUMNS: { TIMESTAMP: 'Timestamp', REQUESTER: 'Name', SONG: 'Song', ARTIST: 'Artist', STATUS: 'Status', VIDEO_ID: 'Video ID', PLAYLIST_ID: 'Playlist ID' }, SEARCH_LIMIT: 5, TIMEZONE: 'America/Chicago' }; // ==================== INITIAL SETUP ==================== /** * Run this ONCE to install the form-submit trigger. * If the trigger already exists, it removes the old one first * so you never get duplicates. */ function initialSetup() { // Remove any existing onFormSubmit triggers to avoid duplicates const triggers = ScriptApp.getProjectTriggers(); triggers.forEach(t => { if (t.getHandlerFunction() === 'onFormSubmit') { ScriptApp.deleteTrigger(t); Logger.log('Removed old onFormSubmit trigger'); } }); // Install a fresh trigger ScriptApp.newTrigger('onFormSubmit') .forSpreadsheet(SpreadsheetApp.getActiveSpreadsheet()) .onFormSubmit() .create(); // Ensure status/video/playlist columns exist ensureColumns(); Logger.log('✅ Setup complete! The script will now run on every form submission.'); Logger.log('Submit a test request to verify it works.'); } /** * Make sure the Status, Video ID, and Playlist ID columns exist. * If they don't, add them as new headers. */ function ensureColumns() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const needed = [CONFIG.COLUMNS.STATUS, CONFIG.COLUMNS.VIDEO_ID, CONFIG.COLUMNS.PLAYLIST_ID]; needed.forEach(col => { if (!headers.includes(col)) { const nextCol = sheet.getLastColumn() + 1; sheet.getRange(1, nextCol).setValue(col); Logger.log('Added missing column: ' + col); } }); } // ==================== MAIN AUTOMATION ==================== /** * Triggered automatically on every form submission. */ function onFormSubmit(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const row = e.range.getRow(); // Read headers and row data const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const rowData = sheet.getRange(row, 1, 1, sheet.getLastColumn()).getValues()[0]; // Map to object const submission = {}; headers.forEach((header, i) => { submission[header] = rowData[i]; }); // Mark as processing setStatus(sheet, row, headers, 'Processing...'); // Process the request processSongRequest(sheet, row, headers, submission); } catch (error) { Logger.log('Error in onFormSubmit: ' + error.toString()); } } /** * Search YouTube for the song, get/create the daily playlist, and add it. */ // Flexible field reader — finds the value whose header CONTAINS any keyword, // so it works no matter how the form/sheet names its columns. function getField(submission, keywords) { if (!submission || typeof submission !== 'object') return ''; for (const key of Object.keys(submission)) { const k = key.toLowerCase(); for (const kw of keywords) { if (k.includes(kw)) { const v = (submission[key] || '').toString().trim(); if (v) return v; } } } return ''; } function processSongRequest(sheet, row, headers, submission) { // Auto-detect song & artist columns regardless of exact header text let song = getField(submission, ['song', 'title', 'track']); let artist = getField(submission, ['artist', 'band', 'singer', 'performer']); if (!song && !artist) { song = getField(submission, ['request', 'what']); } // KEY FIX: the request form ALREADY searched YouTube and saved a Video ID. // Use that directly — don't re-search (which wastes quota and can fail). let videoId = getField(submission, ['video id', 'videoid', 'video_id', 'youtube id']); // Validate it looks like a real YouTube ID (11 chars) if (videoId && !/^[a-zA-Z0-9_-]{11}$/.test(videoId)) videoId = ''; // Only fall back to searching if the form didn't already find one if (!videoId) { if (!song && !artist) { setStatus(sheet, row, headers, 'Skipped — no song info'); return; } const query = artist ? (artist + ' ' + song) : song; videoId = searchYouTube(query); } if (!videoId) { setStatus(sheet, row, headers, 'Song not found'); return; } // Get or create today's playlist const playlist = getOrCreateDailyPlaylist(); // Add video to playlist try { YouTube.PlaylistItems.insert({ snippet: { playlistId: playlist.id, resourceId: { kind: 'youtube#video', videoId: videoId } } }, 'snippet'); // Update sheet with success setStatus(sheet, row, headers, 'Added ✓'); setCell(sheet, row, headers, CONFIG.COLUMNS.VIDEO_ID, videoId); setCell(sheet, row, headers, CONFIG.COLUMNS.PLAYLIST_ID, playlist.id); Logger.log('✅ Added video: ' + videoId + ' (' + (song || artist || 'request') + ')'); } catch (error) { Logger.log('Error adding to playlist: ' + error.toString()); // Handle duplicate — YouTube throws an error if video already in playlist if (error.toString().includes('videoAlreadyInPlaylist')) { setStatus(sheet, row, headers, 'Already in playlist ✓'); setCell(sheet, row, headers, CONFIG.COLUMNS.VIDEO_ID, videoId); setCell(sheet, row, headers, CONFIG.COLUMNS.PLAYLIST_ID, playlist.id); } else { setStatus(sheet, row, headers, 'Error: ' + error.toString().substring(0, 80)); } } } // ==================== YOUTUBE HELPERS ==================== /** * Search YouTube and return the best matching video ID. */ function searchYouTube(query) { try { const results = YouTube.Search.list('snippet', { q: query, type: 'video', maxResults: CONFIG.SEARCH_LIMIT, videoCategoryId: '10' // Music category — improves match quality }); if (results.items && results.items.length > 0) { // Prefer official audio/video — check titles for keywords for (const item of results.items) { const title = (item.snippet.title || '').toLowerCase(); if (title.includes('official') || title.includes('audio') || title.includes('music video')) { return item.id.videoId; } } // Otherwise return the first result return results.items[0].id.videoId; } Logger.log('No YouTube results for: ' + query); return null; } catch (error) { Logger.log('YouTube search error: ' + error.toString()); return null; } } /** * Get today's playlist, or create a new one if it doesn't exist. * Playlist title format: "Song Requests - MM/DD/YY" */ function getOrCreateDailyPlaylist() { const now = new Date(); const dateStr = Utilities.formatDate(now, CONFIG.TIMEZONE, 'MM/dd/yy'); const playlistTitle = CONFIG.PLAYLIST_PREFIX + ' - ' + dateStr; // Check if today's playlist already exists const existing = YouTube.Playlists.list('snippet', { mine: true, maxResults: 25 }); if (existing.items) { for (const pl of existing.items) { if (pl.snippet.title === playlistTitle) { Logger.log('Using existing playlist: ' + playlistTitle); return { id: pl.id, title: playlistTitle }; } } } // Create a new one const newPlaylist = YouTube.Playlists.insert({ snippet: { title: playlistTitle, description: 'DJ Sunny — Song requests for ' + dateStr + '. Auto-generated by PartyQueue.' }, status: { privacyStatus: 'unlisted' // Change to 'public' if you want guests to find it } }, 'snippet,status'); Logger.log('🆕 Created new playlist: ' + playlistTitle); return { id: newPlaylist.id, title: playlistTitle }; } // ==================== SHEET HELPERS ==================== /** * Set the Status column for a given row. */ function setStatus(sheet, row, headers, status) { setCell(sheet, row, headers, CONFIG.COLUMNS.STATUS, status); } /** * Set a specific column value by header name. */ function setCell(sheet, row, headers, columnName, value) { const col = headers.indexOf(columnName); if (col >= 0) { sheet.getRange(row, col + 1).setValue(value); } } // ==================== UTILITIES ==================== /** * Manually process any rows that don't have a status yet. * Useful if the trigger missed some submissions, or to backfill. * Run this from the Apps Script editor: select processAllPending → ▶ Run */ function processAllPending() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const statusCol = headers.indexOf(CONFIG.COLUMNS.STATUS); if (statusCol < 0) { Logger.log('No Status column found. Run initialSetup first.'); return; } const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { const status = (data[i][statusCol] || '').toString().trim(); // Skip only rows that genuinely succeeded — RETRY failed/skipped ones if (status === 'Added ✓' || status === 'Already in playlist ✓') { continue; } const submission = {}; headers.forEach((header, idx) => { submission[header] = data[i][idx]; }); const row = i + 1; // 1-indexed Logger.log('Processing row ' + row + '...'); setStatus(sheet, row, headers, 'Processing...'); processSongRequest(sheet, row, headers, submission); // Small delay to avoid hammering YouTube API Utilities.sleep(1500); } Logger.log('✅ All pending rows processed.'); } /** * Clean up old playlists (older than 7 days). * Run manually when you want to tidy up your YouTube channel. */ function cleanupOldPlaylists() { const cutoff = new Date(); cutoff.setDate(cutoff.getDate() - 7); const playlists = YouTube.Playlists.list('snippet', { mine: true, maxResults: 50 }); if (!playlists.items) return; let deleted = 0; playlists.items.forEach(pl => { if (pl.snippet.title.startsWith(CONFIG.PLAYLIST_PREFIX)) { const created = new Date(pl.snippet.publishedAt); if (created < cutoff) { YouTube.Playlists.remove(pl.id); Logger.log('Deleted old playlist: ' + pl.snippet.title); deleted++; } } }); Logger.log('Cleaned up ' + deleted + ' old playlist(s).'); }
